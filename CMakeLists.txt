cmake_minimum_required(VERSION 3.16)
project(kcf_tool LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
option(KCF_TOOL_BUILD_GUI "Build Qt Widgets application" ON)
option(KCF_TOOL_BUILD_TESTS "Build backend and widget smoke checks" OFF)
set(KCF_SOURCE_DIR "" CACHE PATH "External KCF R1-R5 source root (optional)")

add_library(kcf_tool_backend src/backend/mock_backend.cpp)
target_include_directories(kcf_tool_backend PUBLIC include)
if(KCF_SOURCE_DIR)
    if(NOT EXISTS "${KCF_SOURCE_DIR}/kcf/include/kcf/dynamic/dynamic_service_client.hpp")
        message(FATAL_ERROR "KCF_SOURCE_DIR must provide the R5 public dynamic APIs")
    endif()
    find_package(Threads REQUIRED)
    # Build the external core only, never copy sources or configure its examples.
    add_subdirectory("${KCF_SOURCE_DIR}/kcf" "${CMAKE_CURRENT_BINARY_DIR}/external/kcf" EXCLUDE_FROM_ALL)
    target_sources(kcf_tool_backend PRIVATE src/backend/kcf_backend.cpp src/backend/value_codec.cpp)
    target_link_libraries(kcf_tool_backend PRIVATE kcf_core)
else()
    target_sources(kcf_tool_backend PRIVATE src/backend/kcf_backend_stub.cpp)
endif()

if(KCF_TOOL_BUILD_GUI)
    find_package(Qt6 REQUIRED COMPONENTS Widgets)
    add_executable(kcf_tool src/main.cpp src/ui/main_window.cpp)
    target_link_libraries(kcf_tool PRIVATE kcf_tool_backend Qt6::Widgets)
endif()
if(KCF_TOOL_BUILD_TESTS)
    enable_testing()
    # Preserve Mock and disconnected-stub regression coverage without KCF.
    add_executable(kcf_tool_backend_contract tests/backend_contract.cpp
        src/backend/mock_backend.cpp src/backend/kcf_backend_stub.cpp)
    target_include_directories(kcf_tool_backend_contract PRIVATE include)
    add_test(NAME kcf_tool_backend_contract COMMAND kcf_tool_backend_contract)
    if(KCF_TOOL_BUILD_GUI)
        add_executable(kcf_tool_smoke tests/smoke.cpp src/ui/main_window.cpp
            src/backend/mock_backend.cpp src/backend/kcf_backend_stub.cpp)
        target_include_directories(kcf_tool_smoke PRIVATE include)
        target_link_libraries(kcf_tool_smoke PRIVATE Qt6::Widgets)
        add_test(NAME kcf_tool_smoke COMMAND kcf_tool_smoke)
        set_tests_properties(kcf_tool_smoke PROPERTIES ENVIRONMENT "QT_QPA_PLATFORM=offscreen")
    endif()
endif()

if(KCF_TOOL_BUILD_TESTS AND KCF_SOURCE_DIR)
    add_executable(kcf_tool_real_fixture tests/real_fixture.cpp "${KCF_SOURCE_DIR}/bringup/src/bringup.cpp")
    target_include_directories(kcf_tool_real_fixture PRIVATE "${KCF_SOURCE_DIR}/bringup/include")
    target_link_libraries(kcf_tool_real_fixture PRIVATE kcf_core)
    add_executable(kcf_tool_real_backend tests/real_backend.cpp)
    target_link_libraries(kcf_tool_real_backend PRIVATE kcf_tool_backend)
    add_test(NAME kcf_tool_real_backend COMMAND kcf_tool_real_backend $<TARGET_FILE:kcf_tool_real_fixture>)
    add_executable(kcf_tool_value_codec tests/value_codec.cpp)
    target_include_directories(kcf_tool_value_codec PRIVATE src/backend)
    target_link_libraries(kcf_tool_value_codec PRIVATE kcf_tool_backend kcf_core)
    add_test(NAME kcf_tool_value_codec COMMAND kcf_tool_value_codec)
    # Keep assertions in fixture checks even for Release builds.
    foreach(t kcf_tool_real_fixture kcf_tool_real_backend kcf_tool_value_codec)
        target_compile_options(${t} PRIVATE -UNDEBUG)
    endforeach()
    set_tests_properties(kcf_tool_real_backend PROPERTIES RUN_SERIAL TRUE TIMEOUT 120)
    add_executable(kcf_tool_registration_hardening tests/registration_hardening.cpp)
    target_link_libraries(kcf_tool_registration_hardening PRIVATE kcf_tool_backend kcf_core)
    target_compile_options(kcf_tool_registration_hardening PRIVATE -UNDEBUG)
    target_link_options(kcf_tool_registration_hardening PRIVATE
        "-Wl,--wrap=_ZN3kcf18DynamicTopicReader4OpenERKNSt7__cxx1112basic_stringIcSt11char_traitsIcESaIcEEERKNS_14TypeDescriptorE"
        "-Wl,--wrap=_ZN3kcf22DynamicParameterClient4OpenERKNSt7__cxx1112basic_stringIcSt11char_traitsIcESaIcEEERKNS_14TypeDescriptorE"
        "-Wl,--wrap=_ZN3kcf22DynamicParameterClient3GetERNS_14DynamicPayloadE"
        "-Wl,--wrap=_ZN3kcf20DynamicServiceClient4OpenEttRKNS_14TypeDescriptorES3_jj"
        "-Wl,--wrap=_ZN3kcf19IntrospectionClient13ListEndpointsERKNS_11RuntimeInfoERSt6vectorINS_12EndpointInfoESaIS5_EE"
        "-Wl,--wrap=_ZN3kcf19IntrospectionClient12ListServicesERKNS_11RuntimeInfoERSt6vectorINS_11ServiceInfoESaIS5_EE")
    add_test(NAME kcf_tool_registration_hardening COMMAND kcf_tool_registration_hardening)
    set_tests_properties(kcf_tool_registration_hardening PROPERTIES RUN_SERIAL TRUE TIMEOUT 60)
    if(KCF_TOOL_BUILD_GUI)
        add_executable(kcf_tool_real_smoke tests/real_smoke.cpp src/ui/main_window.cpp)
        target_link_libraries(kcf_tool_real_smoke PRIVATE kcf_tool_backend Qt6::Widgets)
        target_compile_options(kcf_tool_real_smoke PRIVATE -UNDEBUG)
        add_test(NAME kcf_tool_real_smoke COMMAND kcf_tool_real_smoke $<TARGET_FILE:kcf_tool_real_fixture>)
        set_tests_properties(kcf_tool_real_smoke PROPERTIES ENVIRONMENT "QT_QPA_PLATFORM=offscreen" RUN_SERIAL TRUE TIMEOUT 30)
    endif()
endif()

if(KCF_TOOL_BUILD_TESTS AND KCF_SOURCE_DIR AND KCF_TOOL_BUILD_GUI)
    add_executable(kcf_tool_parameter_smoke tests/parameter_smoke.cpp src/ui/main_window.cpp)
    target_link_libraries(kcf_tool_parameter_smoke PRIVATE kcf_tool_backend kcf_core Qt6::Widgets)
